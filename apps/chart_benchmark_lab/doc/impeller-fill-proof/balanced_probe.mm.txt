__attribute__((constructor)) static void installMiddleOut() {
 @autoreleasepool {
  id<MTLDevice> device=MTLCreateSystemDefaultDevice();
  id<MTLCommandQueue> queue=[device newCommandQueue];
  id<MTLCommandBuffer> command=[queue commandBuffer];
  MTLTextureDescriptor* td=[MTLTextureDescriptor texture2DDescriptorWithPixelFormat:MTLPixelFormatBGRA8Unorm width:1 height:1 mipmapped:NO];td.usage=MTLTextureUsageRenderTarget;
  id<MTLTexture> texture=[device newTextureWithDescriptor:td];
  MTLRenderPassDescriptor* pass=[MTLRenderPassDescriptor renderPassDescriptor];pass.colorAttachments[0].texture=texture;pass.colorAttachments[0].loadAction=MTLLoadActionClear;pass.colorAttachments[0].storeAction=MTLStoreActionDontCare;
  id<MTLRenderCommandEncoder> encoder=[command renderCommandEncoderWithDescriptor:pass];
  Class cls=object_getClass(encoder);
  SEL selector=@selector(drawIndexedPrimitives:indexCount:indexType:indexBuffer:indexBufferOffset:);
  Method method=class_getInstanceMethod(cls,selector);IMP original=method_getImplementation(method);
  IMP hook=imp_implementationWithBlock(^(id<MTLRenderCommandEncoder> self,MTLPrimitiveType type,NSUInteger count,MTLIndexType indexType,id<MTLBuffer> buffer,NSUInteger offset){
   auto draw=[&](id<MTLBuffer> indices,NSUInteger n,NSUInteger o){((void(*)(id,SEL,MTLPrimitiveType,NSUInteger,MTLIndexType,id<MTLBuffer>,NSUInteger))original)(self,selector,type,n,indexType,indices,o);};
   if(type!=MTLPrimitiveTypeTriangleStrip || indexType!=MTLIndexTypeUInt32 || count<1024 || offset>buffer.length || count>(buffer.length-offset)/4 || !buffer.contents){draw(buffer,count,offset);return;}
   const uint32_t* source=(const uint32_t*)((const uint8_t*)buffer.contents+offset);
   std::vector<uint32_t> replacement;replacement.reserve(count*2);bool changed=false;bool valid=true;
   for(size_t pos=0;pos<count;){
    size_t finish=pos;while(finish<count && source[finish]!=UINT32_MAX)finish++;
    const size_t n=finish-pos;
    if(n>=1024){
     size_t first=source[pos],last=first+n-1,j=pos+1,a=first+1,b=last;
     while(a<b && valid){if(source[j++]!=a++ || source[j++]!=b--)valid=false;}
     if(a==b && valid && source[j++]!=a)valid=false;
     if(!valid || j!=finish){valid=false;break;}
     EmitBalancedStrip(first,last,[&](size_t a){replacement.push_back((uint32_t)a);},[&](){replacement.push_back(UINT32_MAX);});
     changed=true;
    }else{replacement.insert(replacement.end(),source+pos,source+finish);replacement.push_back(UINT32_MAX);}
    pos=finish+1;
   }
   if(!valid || !changed){draw(buffer,count,offset);return;}
   id<MTLBuffer> rewritten=[device newBufferWithBytes:replacement.data() length:replacement.size()*4 options:MTLResourceStorageModeShared];
   if(!rewritten){draw(buffer,count,offset);return;}
   static size_t rewrites=0;if(++rewrites<=4)fprintf(stderr,"MIDDLE_OUT_REWRITE old=%lu new=%zu\n",(unsigned long)count,replacement.size());
   draw(rewritten,replacement.size(),0);
  });
  class_replaceMethod(cls,selector,hook,method_getTypeEncoding(method));
  [encoder endEncoding];
  fprintf(stderr,"MIDDLE_OUT_PROBE installed %s\n",class_getName(cls));
 }
}
