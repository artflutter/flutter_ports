#pragma once
#include <cstddef>
#include <cstdint>
namespace impeller {
struct Point { float x,y; };
struct PathTessellator { struct VertexWriter { virtual ~VertexWriter()=default;virtual void Write(Point)=0;virtual void EndContour()=0; }; };
}
template <typename IndexT>
class StripPathVertexWriter : public impeller::PathTessellator::VertexWriter {
 public:
  explicit StripPathVertexWriter(impeller::Point* point_buffer,
                                 IndexT* index_buffer,
                                 bool use_balanced_strips)
      : point_buffer_(point_buffer),
        index_buffer_(index_buffer),
        use_balanced_strips_(use_balanced_strips) {}

  ~StripPathVertexWriter() = default;

  size_t GetIndexCount() const { return index_count_; }
  size_t GetPointCount() const { return count_; }

  void EndContour() override {
    if (count_ == 0u || contour_start_ == count_ - 1) {
      // Empty or first contour.
      return;
    }

    // Keep short contours on the compact legacy strip. Large non-convex
    // contours use local strips linked by signed middle-out triangles.
    if (use_balanced_strips_ && count_ - contour_start_ >= 1024u) {
      WriteBalancedStrip(contour_start_, count_ - 1);
      contour_start_ = count_;
      return;
    }

    size_t start = contour_start_;
    size_t end = count_ - 1;

    index_buffer_[index_count_++] = start;

    size_t a = start + 1;
    size_t b = end;
    while (a < b) {
      index_buffer_[index_count_++] = a;
      index_buffer_[index_count_++] = b;
      a++;
      b--;
    }
    if (a == b) {
      index_buffer_[index_count_++] = a;
    }

    contour_start_ = count_;
    index_buffer_[index_count_++] = static_cast<IndexT>(-1);
  }

  void Write(impeller::Point point) override {
    point_buffer_[count_++] = point;
  }

 private:
  void WriteBalancedStrip(size_t first, size_t last) {
    if (last - first < 2u) {
      return;
    }
    if (last - first <= 32u) {
      index_buffer_[index_count_++] = first;
      size_t a = first + 1;
      size_t b = last;
      while (a < b) {
        index_buffer_[index_count_++] = a++;
        index_buffer_[index_count_++] = b--;
      }
      if (a == b) {
        index_buffer_[index_count_++] = a;
      }
      index_buffer_[index_count_++] = static_cast<IndexT>(-1);
      return;
    }

    const size_t middle = first + (last - first) / 2;
    WriteBalancedStrip(first, middle);
    // Every internal diagonal appears with opposite directions in the child
    // strip and this triangle, preserving nonzero and even-odd winding.
    index_buffer_[index_count_++] = first;
    index_buffer_[index_count_++] = middle;
    index_buffer_[index_count_++] = last;
    index_buffer_[index_count_++] = static_cast<IndexT>(-1);
    WriteBalancedStrip(middle, last);
  }

  size_t count_ = 0;
  size_t index_count_ = 0;
  size_t contour_start_ = 0;
  impeller::Point* point_buffer_ = nullptr;
  IndexT* index_buffer_ = nullptr;
  bool use_balanced_strips_ = false;
};
