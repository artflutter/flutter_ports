ChartBenchCase makeCase(String name) => ChartBenchCase(
  mode: 'architecture', name: name, call: ChartBenchCall.frame,
  scene: (n) {
    registerGalleryComponents();
    final field = StressPointField(n);
    final scene = StressChartScene(StressChartType.bar, n, field);
    final config = stressChartConfig(scene: scene, scriptable: false,
      options: chartBenchOptions);
    List<Object?>? prepared;
    if (name.startsWith('prepared')) {
      prepared = [for (var i=0;i<n;i++)
        name == 'prepared_native'
          ? ParsedItem.of(AxisKey.x,'x',i.toDouble(),AxisKey.y,'y',scene.data[i])
          : <String,Object?>{'x':i.toDouble(),'y':scene.data[i]}];
      (config['options']! as Map<String,Object?>)['parsing'] = false;
      final data=config['data']! as Map<String,Object?>;
      ((data['datasets']! as List).first as Map)['data']=prepared;
    }
    return ChartBenchScene(config: config, frame: (chart) {
      if (name != 'unchanged_update') {
        field.step(n, chartBenchStep);
        scene.write(field);
      }
      if (prepared != null) {
        for (var i=0;i<n;i++) {
          final item=prepared![i];
          if (item is ParsedItem) { item.y=scene.data[i]; }
          else { (item! as Map)['y']=scene.data[i]; }
        }
      }
      if (name != 'values_only') {
        chart.update('none');
        return;
      }
      // Restricted experiment: fixed axes/layout/styles, positive values.
      // Omits parsing/update hooks and parsed-data synchronization.
      // This is not a production update implementation.
      final meta = chart.getDatasetMeta(0);
      final scale = meta.vScale!;
      for (var i = 0; i < n; i++) {
        final bar = meta.data[i] as BarElement;
        final head = scale.getPixelForValue(scene.data[i]);
        bar.y = head;
        bar.height = (bar.base - head).abs();
      }
      chart.draw();
    });
  },
);
Object geometry(Chart chart) => [
  for (final e in chart.getDatasetMeta(0).data)
    [for (final key in ['x','y','base','width','height'])
      (e as types.ElementPropertyRead)[key]],
];
Future<void> run() async {
  await Future<void>.delayed(const Duration(seconds: 1));
  animator.refresh = () {};
  final env = Platform.environment;
  final file = File('/tmp/architecture-prepared-clean-${env['ARCH_RUN'] ?? 'a'}.jsonl');
  var cases = ['full_frame','prepared_map','prepared_native'];
  if (env['ARCH_REVERSE'] == '1') cases = cases.reversed.toList();
  for (final n in [20000]) {
    for (final name in cases) {
      var verified = false;
      final Map<String,Object?> result;
      if (name == 'native_chunk') {
        result = nativeCase(n, name);
      } else {
        result = runChartBenchCase(name == 'draw' ? old.makeCase(name) : makeCase(name),n,
          const ChartBenchSettings(batches: 7,batchMs: 200,warmupMs: 800),
          onChart: !name.startsWith('prepared') ? null : (chart) {
            final before = jsonEncode(geometry(chart));
            final data=chart.data as Map;
            final dataset=(data['datasets'] as List).first as Map;
            final current=dataset['data'] as List;
            final referenceScene=StressChartScene(StressChartType.bar,n,StressPointField(n));
            for(var i=0;i<n;i++) {
              final item=current[i];
              referenceScene.data[i]=item is ParsedItem ? item.y : (item as Map)['y'];
            }
            final canvas=UiCanvasElement(width:800,height:480);
            final reference=Chart(canvas,stressChartConfig(scene:referenceScene,
              scriptable:false,options:chartBenchOptions));
            try { verified=before==jsonEncode(geometry(reference)); }
            finally { reference.destroy();canvas.getContext2d().dispose(); }
            if (!verified) throw StateError('Geometry mismatch at N=$n');
          });
      }
      file.writeAsStringSync('${jsonEncode({...result, if(name.startsWith('prepared')) 'allBarGeometryMatchesFullUpdate': verified})}\n',mode: FileMode.append,flush: true);
      print('ARCH $n $name');
    }
  }
  exit(0);
}
void main() {
  WidgetsFlutterBinding.ensureInitialized();
  runApp(const MaterialApp(home: Scaffold(body: Text('Architecture diagnosis'))));
  unawaited(run().catchError((Object e, StackTrace st) { print('$e\n$st'); exit(1); }));
}
