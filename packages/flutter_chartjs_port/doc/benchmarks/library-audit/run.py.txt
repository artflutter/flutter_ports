import os,sys,subprocess,re,time,json,gzip
from pathlib import Path
root=Path(__file__).resolve().parents[2]
out=root/'.dart_tool/library_audit'
mode=sys.argv[1]
app='/tmp/chart-library-audit.app/Contents/MacOS/chartjs_gallery'
def launch(label, env, profile=False):
 with (out/(label+'.log')).open('w') as log:
  p=subprocess.Popen([app],env={**os.environ,**env},stdout=log,stderr=subprocess.STDOUT)
  try:
   if profile:
    for _ in range(100):
     t=(out/(label+'.log')).read_text(); m=re.search(r'http://127.0.0.1:(\d+)/(\S+)',t)
     if m: break
     time.sleep(.1)
    else: raise RuntimeError(t)
    subprocess.run(['dart','--packages='+str(root/'.dart_tool/package_config.json'),str(out/'capture.dart'),f'ws://127.0.0.1:{m[1]}/{m[2]}ws',str(out/label)],check=True)
    raw=(out/(label+'.json')).read_bytes()
    (out/(label+'.json.gz')).write_bytes(gzip.compress(raw))
    (out/(label+'.json')).unlink()
   else:
    p.wait(timeout=240)
    if p.returncode: raise RuntimeError((out/(label+'.log')).read_text())
    rows=[json.loads(l.split('AUDIT ',1)[1]) for l in (out/(label+'.log')).read_text().splitlines() if 'AUDIT {' in l]
    (out/(label+'.json')).write_text(json.dumps(rows,indent=2))
    print(label,len(rows),flush=True)
  finally:
   if p.poll() is None: p.terminate();p.wait(timeout=10)
if mode=='measure':
 for r in ['forward','reverse']:
  launch(r,{'N':'1000,5000','REVERSE':str(int(r=='reverse')),'CASES':'chart,chart.c.indexable,chart.line.primitive','CALL':'update'})
else:
 for label,case,type_ in [('static','chart.b','line'),('indexed','chart.c.indexable','line'),('scriptable','chart.c','line'),('bubble','chart.bubble','bubble'),('bar','chart.bar','bar')]:
  launch('profile-'+label,{'N':'5000','CASES':case,'TYPE':type_,'CALL':'update','PROFILE':'1'},True)
