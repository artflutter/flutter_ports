import json,statistics
from pathlib import Path
p=Path(__file__).resolve().parent;full=lambda r:sum(float(r[k]) for k in ['updateMedianMs','drawMedianMs','endMedianMs'])
rows=[];checks=0;cells=0
for prefix in ['guard','one']:
 runs={k:json.loads((p/(prefix+'-'+k+'.json')).read_text()) for k in ['a1','b1','b2','a2']}
 cells+=sum(len(v) for v in runs.values())
 for x in runs['b1']:
  key=x['case'].rsplit('.',1)[0]+'.baseline'
  a=[next(v for v in runs[k] if v['case']==key) for k in ['a1','a2']]
  b=[next(v for v in runs[k] if v['case']==x['case']) for k in ['b1','b2']]
  counts=[]
  for r in b:
   assert r['referenceMatches'] and r['pathMatches'] and r['pixelByteMismatches']==0
   if 'preflight' in r['variant'] or r['variant']=='points_one_scan':
    initial=2 if r['variant']=='points_one_scan' else 4
    n=(r['preflightScans']-initial)/r['frames'];assert n==(1 if r['variant']=='points_one_scan' else 4)
    assert r['pointHits']>0 and r['contextHits']>0
    counts.append(n)
   checks+=1
  fa=statistics.mean(map(full,a));fb=statistics.mean(map(full,b))
  row={'round':prefix,'case':x['case'],'n':5000,'baselineFullMs':fa,'candidateFullMs':fb,'fullGainPercent':100*(1-fb/fa),'pairFullGains':[100*(1-full(y)/full(z)) for y,z in zip(b,a)],'preflightScansPerUpdate':counts}
  rows.append(row);print(prefix,x['case'],round(fa,3),'->',round(fb,3),round(row['fullGainPercent'],1),'pairs',[round(x,1) for x in row['pairFullGains']])
(p/'guard-summary.json').write_text(json.dumps(rows,indent=2)+'\n')
v=json.loads((p/'validation.json').read_text());v['supplementalTimingCells']=cells;v['supplementalDifferentialChecks']=checks;v['scansPerUpdateVerified']=[1,4];(p/'validation.json').write_text(json.dumps(v,indent=2)+'\n')
