import os,json,subprocess
from pathlib import Path
out=Path(__file__).resolve().parent
runs={}
for app in ['baseline','production']:
 directory=out/('pixels-'+app);directory.mkdir(exist_ok=True)
 with (out/('pixels-'+app+'.log')).open('w') as log:
  subprocess.run([f'/tmp/path-chunks-{app}.app/Contents/MacOS/chartjs_gallery','-ApplePersistenceIgnoreState','YES'],env={**os.environ,'PIXELS':'1','FRAMES':'0','OUTDIR':str(directory)},stdout=log,stderr=subprocess.STDOUT,timeout=180,check=True)
 rows=json.loads((directory/'results.json').read_text());assert len(rows)==300,len(rows);runs[app]={r['scene']:r for r in rows};print(app,len(rows),flush=True)
assert runs['baseline'].keys()==runs['production'].keys()
differences=[]
for name,a in runs['baseline'].items():
 b=runs['production'][name]
 if a['sha256']!=b['sha256']:
  aa=(out/'pixels-baseline'/(name+'.rgba')).read_bytes();bb=(out/'pixels-production'/(name+'.rgba')).read_bytes();d=[abs(x-y) for x,y in zip(aa,bb)]
  differences.append({'scene':name,'changedBytes':sum(x!=0 for x in d),'maxDelta':max(d)})
(out/'pixel-comparison.json').write_text(json.dumps({'comparisons':300,'differences':differences,'runs':runs},indent=2)+'\n')
print('differences',differences,flush=True)
assert not differences
