for call in ['update','animate']:
 rows={tag:next(r for r in records if r['call']==call) for tag,records in repeat.items()};a=statistics.mean(rows[k]['buildMs'] for k in ['a1','a2']);c=statistics.mean(rows[k]['buildMs'] for k in ['b1','b2']);repeats.append({'call':call,'beforeMeanUiMs':a,'afterMeanUiMs':c,'reductionPct':100*(1-c/a),'runs':rows})

for name,r in gc.items():

 p90=sorted(f['uiMs'] for f in rows)[math.ceil(len(rows)*.9)-1];slow=[f for f in rows if f['uiMs']>=p90]

summary={'matrixComplete':True,'gcComplete':True,'production':prod,'modeSummary':summaries,'heavyRepeat':repeats,'gc':gc,'worstByFps':sorted(prod,key=lambda r:r['fps'])[:8],'minReadouts':min(r['samples'] for r in allframes),'minTimings':min(r['frameTimings'] for r in allframes),'totalTimings':sum(r['frameTimings'] for r in allframes)}
(b/'summary.json').write_text(json.dumps(summary,indent=2)+'\n');(b/'visibility.json').write_text(json.dumps(vis,indent=2)+'\n')
studio=json.loads((w.parent/'studio-summary.json').read_text());(out/'studio-summary.json').write_text(json.dumps(studio,indent=2)+'\n');(out/'summary.json').write_text(json.dumps({'studio':studio,'builtin':summary,'matrixComplete':True,'gcComplete':True},indent=2)+'\n')
