import 'dart:async';
import 'dart:convert';
import 'dart:io';
import 'dart:math' as math;
import 'dart:ui' as ui;
import 'package:crypto/crypto.dart';
import 'package:flutter_chartjs_port/src/canvas/canvas.dart';
import 'package:flutter_chartjs_port/src/canvas/canvas_ui.dart';
import 'package:flutter_chartjs_port/src/elements/element.point.dart';
import 'package:flutter_chartjs_port/src/elements/element_options.dart';

Future<void> runPixels() async {
 final directory=Directory(Platform.environment['OUTDIR']!)..createSync(recursive:true);
 final rows=<Map<String,Object?>>[];
 await Future<void>.delayed(const Duration(seconds:2));
 Future<void> render(String name,int dpr,void Function(UiCanvasContext) draw) async {
  final element=UiCanvasElement(width:160*dpr,height:160*dpr);
  final ctx=element.getContext2d();ctx.scale(dpr.toDouble(),dpr.toDouble());draw(ctx);
  final recorder=ui.PictureRecorder();ctx.paint(ui.Canvas(recorder),ui.Offset.zero,1,1);
  final picture=recorder.endRecording();final image=picture.toImageSync(160*dpr,160*dpr);
  final bytes=(await image.toByteData(format:ui.ImageByteFormat.rawRgba))!.buffer.asUint8List();
  File('${directory.path}/$name.rgba').writeAsBytesSync(bytes);
  rows.add({'scene':name,'bytes':bytes.length,'sha256':sha256.convert(bytes).toString()});
  image.dispose();picture.dispose();ctx.dispose();
 }
 for(final dpr in [1,2]) for(final radius in [1.0,3.0,12.0]) for(final width in [.5,1.0,3.0]) for(final alpha in [64,255]) for(final indexed in [false,true]) for(final transform in [0,1,2]){
  await render('circle-$dpr-$radius-$width-$alpha-$indexed-$transform',dpr,(ctx){
   if(transform==1){ctx.translate(20,10);ctx.rotate(.3);}if(transform==2)ctx.scale(1.2,.8);
   for(var i=0;i<600;i++){
    final p=PointElement()..x=-5+(i*7.371)%170..y=-3+(i*5.127)%166;
    p.options=PointOptions({'radius':radius,'hoverRadius':radius,'borderWidth':width,'pointStyle':'circle','rotation':0.0,'backgroundColor':ui.Color.fromARGB(alpha,indexed&&i.isOdd?255:30,90,200),'borderColor':const ui.Color(0xff2499ff)});p.draw(ctx);
   }
   ctx.fill();ctx.stroke();
  });
 }
 for(final dpr in [1,2]) for(final transform in [0,1,2]) for(final mode in ['open','closed','mixed','dash','clip','mutate','long']) for(final exposed in [false,true]){
  await render('path-$dpr-$transform-$mode-$exposed',dpr,(ctx){
   if(transform==1){ctx.translate(12,5);ctx.rotate(.2);}if(transform==2)ctx.scale(1.1,.8);
   if(mode=='clip'){ctx.beginPath();ctx.rect(8,10,120,100);ctx.clip();}
   ctx.fillStyle=const ui.Color(0x203399ff);ctx.strokeStyle=const ui.Color(0x80992244);ctx.lineWidth=.75;
   if(mode=='dash')ctx.setLineDash([2,3]);
   final path=Path2D();if(exposed)path.commands.clear();
   path.moveTo(-0.0,12);
   for(var i=0;i<(mode=='long'?20000:512);i++){
    path.lineTo(4+(i*17.0%150),5+(i*13.0%149));
    if(i%1023==0){path.closePath();path.moveTo(8,9);}
   }
   if(mode=='closed')path.closePath();
   if(mode=='mixed'){path.bezierCurveTo(20,30,40,70,90,100);path.arc(50,60,20,0,math.pi);path.roundRect(15,20,30,45,[2,4]);}
   if(mode!='long')ctx.fill(path);ctx.stroke(path);
   if(mode=='mutate'){path.commands[1]=const LineToCommand(3,6);ctx.stroke(path);path.lineTo(5,7);ctx.stroke(path);}
  });
 }
 File('${directory.path}/results.json').writeAsStringSync(jsonEncode(rows));
 print('PIXELS ${rows.length}');exit(0);
}
