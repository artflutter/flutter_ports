import os,json,subprocess
from pathlib import Path
out=Path(__file__).resolve().parent

def run(label,app,extra,expected):
 saved=out/(label+'.json')
 if saved.exists():return json.loads(saved.read_text())
 print('START',label,flush=True)
 result=out/(label+'.results');result.unlink(missing_ok=True);Path(str(result)+'.window').unlink(missing_ok=True)
 env={**os.environ,'FRAMES':'1','PIXELS':'0','N':'20000','BENCH_DISPLAY_ID':'5','CALLS':'update','STYLES':'scriptable','CONNECTED':'0,1','LABELS':'0,50,200','SPIN':'0','SECONDS':'6','SHAPE':'plain',**extra,'OUTPUT':str(result)}
 with (out/(label+'.log')).open('w') as log:
  subprocess.run([f'/tmp/path-chunks-{app}.app/Contents/MacOS/chartjs_gallery','-ApplePersistenceIgnoreState','YES'],env=env,stdout=log,stderr=subprocess.STDOUT,timeout=180,check=True)
 rows=[json.loads(l.split(' ',1)[1]) for l in result.read_text().splitlines() if l.startswith('LINE_FRAME ')]
 assert len(rows)==expected and all(r['samples']>=6 and r['frameTimings']>=50 and r['canvas']=='1036.0x520.0@2.0' and r['items']==int(env['N']) for r in rows),(label,'invalid timings')
 vis=[json.loads(l) for l in Path(str(result)+'.window').read_text().splitlines()];prep=next(r for r in vis if r['event']=='prepared')
 assert (prep['width'],prep['height'],prep['dpr'],prep['displayId'],prep['maxFps'])==(1100,900,2,5,60),(label,prep)
 assert not any(r['time']>prep['time']+1 and (r['occluded'] or r['appHidden'] or r['miniaturized'] or (r['width'],r['height'],r['dpr'],r['displayId'],r['maxFps'])!=(1100,900,2,5,60)) for r in vis),(label,'window changed')
 saved.write_text(json.dumps(rows,indent=2)+'\n');print('DONE',label,len(rows),flush=True);return rows

long={}
for tag,app in [('a1','baseline'),('b1','production'),('b2','production'),('a2','baseline')]:
 long[tag]=run('long-'+tag,app,{'CALLS':'update','CONNECTED':'1','LABELS':'200','SPIN':'1','SECONDS':'30'},1)
(out/'long-repeat.json').write_text(json.dumps(long,indent=2)+'\n')
