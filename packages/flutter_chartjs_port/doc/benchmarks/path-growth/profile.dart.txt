import 'dart:async';
import 'dart:convert';
import 'dart:io';
import 'package:vm_service/vm_service_io.dart';
Future<void> main(List<String> args) async {
 final service=await vmServiceConnectUri(args[0]);
 final path=args[1];
 final gcOnly=Platform.environment['GC_ONLY']=='1';
 await service.setFlag('profiler',gcOnly?'false':'true');
 final vm=await service.getVM();
 final isolate=vm.isolates!.firstWhere((i)=>i.name=='main');
 final events=<Object?>[];
 await service.streamListen('GC');
 final sub=service.onGCEvent.listen((e)=>events.add(e.toJson()));
 await service.setVMTimelineFlags(gcOnly?['Dart','GC']:['Dart','GC','Embedder']);
 await Future<void>.delayed(const Duration(seconds:4));
 await service.clearCpuSamples(isolate.id!);
 await service.clearVMTimeline();
 final start=await service.getVMTimelineMicros();
 try {final a=await service.getAllocationProfile(isolate.id!,reset:true);await File('$path.heap-start').writeAsString(jsonEncode(a.toJson()));}catch(e){print('heap start: $e');}
 await Future<void>.delayed(const Duration(seconds:16));
 final end=await service.getVMTimelineMicros();
 final samples=gcOnly?null:await service.getCpuSamples(isolate.id!,start.timestamp!,end.timestamp!-start.timestamp!);
 await File(path).writeAsString(jsonEncode(samples?.toJson()??{'profilerDisabled':true,'sampleCount':0,'samplePeriod':0,'timeOriginMicros':start.timestamp,'timeExtentMicros':end.timestamp!-start.timestamp!,'functions':[],'samples':[]}));
 try {final t=await service.getVMTimeline(timeOriginMicros:start.timestamp,timeExtentMicros:end.timestamp!-start.timestamp!);await File('$path.timeline').writeAsString(jsonEncode(t.toJson()));}catch(e){print('timeline: $e');}
 try {final a=await service.getAllocationProfile(isolate.id!);await File('$path.heap-end').writeAsString(jsonEncode(a.toJson()));}catch(e){print('heap end: $e');}
 await File('$path.gc').writeAsString(jsonEncode(events));
 await File('$path.window').writeAsString(jsonEncode({'start':start.timestamp,'end':end.timestamp}));
 print('samples: ${samples?.sampleCount ?? 0}, GC events: ${events.length}');
 try {
  final hp=await service.getAllocationProfile(isolate.id!);
  final cls=hp.members!.firstWhere((c)=>c.classRef?.name=='Path2D').classRef!;
  final instances=await service.getInstances(isolate.id!,cls.id!,8);
  final inspected=<Object?>[];
  for(final ref in instances.instances!){
   try {
    final obj=(await service.getObject(isolate.id!,ref.id!)).toJson();
    final fields=obj['fields'] as List?;if(fields==null)continue;
    final row=<String,Object?>{};
    for(final f in fields){
     final name=f['decl']['name'] as String;
     if(!['_length','_chunkLength','_initialCapacityHint','_coordinates','_kinds','_kindChunks','_coordinateChunks','_commands'].contains(name))continue;
     final value=f['value'] as Map;
     row[name]={'kind':value['kind'],'value':value['valueAsString'],'length':value['length']};
     if(value['id']!=null && ['TypedData','List'].contains(value['kind'])){
      final child=(await service.getObject(isolate.id!,value['id'] as String)).toJson();
      row[name]={'kind':child['kind'],'length':child['length']};
     }
    }
    inspected.add(row);
   }catch(e){print('path instance: $e');}
  }
  await File('$path.paths').writeAsString(jsonEncode(inspected));
 }catch(e){print('path inspection: $e');}
 await sub.cancel();await service.dispose();
}
