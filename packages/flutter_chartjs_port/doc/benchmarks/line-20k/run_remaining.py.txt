import json,os,subprocess
from pathlib import Path
root=Path(__file__).resolve().parents[2];out=root/'.dart_tool/line_20k'
def read(path,marker):return [json.loads(l.split(marker,1)[1]) for l in path.read_text().splitlines() if marker+'{' in l]
def run(label,variant,extra,expected,frames=False):
 print('START',label,flush=True)
 result=out/(label+'.results');result.unlink(missing_ok=True);logpath=out/(label+'.log')
 env={**os.environ,'FRAMES':'0','CALLS':'draw,update,data,animate','STYLES':'static,indexed,scriptable','CONNECTED':'0,1','LABELS':'0','SPIN':'0','SECONDS':'5',**extra,'OUTPUT':str(result)}
 with logpath.open('w') as log:
  subprocess.run([f'/tmp/line-20k-{variant}.app/Contents/MacOS/chartjs_gallery','-ApplePersistenceIgnoreState','YES'],env=env,stdout=log,stderr=subprocess.STDOUT,check=True,timeout=240)
 marker='LINE_FRAME ' if frames else 'LINE_CPU '
 rows=read(result if result.exists() else logpath,marker)
 if len(rows)!=expected:raise RuntimeError((label,len(rows),expected))
 if frames and any(r['samples']<4 or r['frameTimings']<25 or r['canvas']!='1036.0x520.0@2.0' for r in rows):raise RuntimeError((label,'invalid frame samples',rows))
 if result.exists():
  header='\n'.join(l for l in logpath.read_text().splitlines() if 'LINE_FRAME {' not in l and 'LINE_CPU {' not in l)
  logpath.write_text(header+'\n'+result.read_text())
 (out/(label+'.json')).write_text(json.dumps(rows,indent=2)+'\n');print('DONE',label,len(rows),flush=True);return rows
original=out/'frames-initial.results'
if not original.exists():original.write_bytes((out/'frames-final.results').read_bytes())
rows=read(original,'LINE_FRAME ')
for index,r in enumerate(rows):
 if r['samples']>=4 and r['frameTimings']>=25 and r['canvas']=='1036.0x520.0@2.0':continue
 name=f'frame-repair-{index}'
 saved=out/(name+'.json')
 if saved.exists():rows[index]=json.loads(saved.read_text())[0];continue
 rows[index]=run(name,'final',{'FRAMES':'1','CALLS':r['call'],'STYLES':r['style'],'CONNECTED':'1' if r['connected'] else '0','LABELS':str(r['labels'])},1,True)[0]
(out/'frames-final.log').write_text(''.join('LINE_FRAME '+json.dumps(r)+'\n' for r in rows))
(out/'frames-final.json').write_text(json.dumps(rows,indent=2)+'\n')
run('frames-spin','final',{'FRAMES':'1','LABELS':'200','SPIN':'1'},24,True)
run('cpu-final2','final',{},12)
run('cpu-a2','a',{},12)
