import os,subprocess,json
from pathlib import Path
for name,variant in [('a1','a'),('b1','b'),('b2','b'),('a2','a')]:
 for group,ns,cases in [('primary','1000,5000,20000','baseline,draw,anim'),('controls','1000,5000','update_line,draw_line,update_bubble,draw_doughnut,thickness')]:
  run='impl-'+name+'-'+group
  path=Path('/tmp/bar-'+run+'.jsonl')
  if path.exists():raise RuntimeError('Result exists: '+str(path))
  with open('/tmp/bar-'+run+'.log','w') as log:
   subprocess.run(['/tmp/bar-implementation-'+variant+'.app/Contents/MacOS/chartjs_gallery'],env={**os.environ,'BAR_RUN':run,'BAR_VARIANTS':cases,'BAR_N':ns},stdout=log,stderr=subprocess.STDOUT,check=True)
  print(run, len(path.read_text().splitlines()),flush=True)
