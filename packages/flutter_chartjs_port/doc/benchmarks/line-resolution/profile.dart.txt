import 'dart:convert';
import 'dart:io';
import 'package:vm_service/vm_service_io.dart';
Future<void> main(List<String> args) async {
 final service=await vmServiceConnectUri(args[0]);
 await service.setFlag('profiler','true');
 final vm=await service.getVM();
 final isolate=vm.isolates!.firstWhere((i)=>i.name=='main');
 await Future<void>.delayed(const Duration(seconds:2));
 await service.clearCpuSamples(isolate.id!);
 try { final a=await service.getAllocationProfile(isolate.id!, reset:true, gc:true); await File(args[1]+'.allocation-start').writeAsString(jsonEncode(a.toJson())); } catch(e) { print('allocation: $e'); }
 await Future<void>.delayed(const Duration(seconds:12));
 final samples=await service.getCpuSamples(isolate.id!,0,1<<62);
 await File(args[1]).writeAsString(jsonEncode(samples.toJson()));
 print('samples: ${samples.sampleCount}');
 try { final a=await service.getAllocationProfile(isolate.id!); await File(args[1]+'.allocation-end').writeAsString(jsonEncode(a.toJson())); } catch(e) { print('allocation: $e'); }
 await service.dispose();
}
