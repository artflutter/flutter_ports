class _ReferenceChart extends Chart {
  _ReferenceChart(super.item, super.userConfig);
  @override
  bool get supportsNativeBarIndexReads => false;
}

class _ReadMap extends MapBase<String, Object?> {
  _ReadMap(this.data, this.reads);
  final Map<String, Object?> data;
  final List<String> reads;
  @override
  Object? operator [](Object? key) {
    reads.add('$key');
    return data[key];
  }

  @override
  void operator []=(String key, Object? value) => data[key] = value;
  @override
  Iterable<String> get keys => data.keys;
  @override
  void clear() => data.clear();
  @override
  Object? remove(Object? key) => data.remove(key);
}

class _ReadList extends ListBase<Object?> {
  int reads = 0;
  @override
  int get length {
    reads++;
    return 1;
  }

  @override
  set length(int value) => throw UnsupportedError('length');
  @override
  Object? operator [](int index) {
    reads++;
    return <String, Object?>{};
  }

  @override
  void operator []=(int index, Object? value) =>
      throw UnsupportedError('write');
}

class _ObservedCategory extends CategoryScale {
  _ObservedCategory(super.cfg);
  int matches = 0;
  static final value = JsClass(
    'ObservedBarCategory',
    superclass: CategoryScale.jsClass,
    statics: {'id': 'observed-bar-category'},
    construct: (_, args) => _ObservedCategory(scaleCfgArg(args)),
  );
  @override
  List<types.ChartMeta> getMatchingVisibleMetas([String? type]) {
    matches++;
    return super.getMatchingVisibleMetas(type);
  }
}

class _ValuesController extends BarController {
  _ValuesController(super.chart, super.datasetIndex, super.newTarget);
  static List<double> values = [];
  static final value = JsClass(
    'BarValuesProbe',
    superclass: BarController.jsClass,
    statics: {'id': 'bar-values-probe'},
    construct: (target, args) {
      final (:chart, :datasetIndex) = controllerArgs(args);
      return _ValuesController(chart, datasetIndex, target);
    },
  );
  @override
  List<double> getAllParsedValues(types.Scale scale) => values;
}

Map<String, Object?> _config({
  bool horizontal = false,
  bool skipNull = false,
}) => {
  'type': 'bar',
  'data': <String, Object?>{
    'labels': <String>['a', 'b', 'c', 'd'],
    'datasets': <Object?>[
      <String, Object?>{
        'data': <double>[1, 2, 3, 4],
        'skipNull': skipNull,
        'backgroundColor': const ui.Color(0x803366AA),
        'borderWidth': 1.0,
      },
    ],
  },
  'options': <String, Object?>{
    'responsive': false,
    'animation': false,
    'indexAxis': horizontal ? 'y' : 'x',
    'plugins': <String, Object?>{'legend': false, 'tooltip': false},
    'scales': <String, Object?>{
      'x': <String, Object?>{
        'type': horizontal ? 'linear' : 'category',
        'display': false,
        if (horizontal) ...{'min': 0.0, 'max': 10.0},
      },
      'y': <String, Object?>{
        'type': horizontal ? 'category' : 'linear',
        'display': false,
        if (!horizontal) ...{'min': 0.0, 'max': 10.0},
      },
    },
  },
};
List<Object?> _datasets(Map<String, Object?> cfg) =>
    (cfg['data']! as Map<String, Object?>)['datasets']! as List<Object?>;
Map<String, Object?> _dataset(Map<String, Object?> cfg) =>
    _datasets(cfg)[0]! as Map<String, Object?>;
Map<String, Object?> _options(Map<String, Object?> cfg) =>
    cfg['options']! as Map<String, Object?>;
Map<String, Object?> _scales(Map<String, Object?> cfg) =>
    _options(cfg)['scales']! as Map<String, Object?>;

Chart _chart(Object config, {bool reference = false}) {
  final canvas = UiCanvasElement(width: 640, height: 400);
  final chart = reference
      ? _ReferenceChart(canvas, config)
      : Chart(canvas, config);
  addTearDown(chart.destroy);
  return chart;
}

bool _eligible(Chart chart) {
  final meta = chart.getDatasetMeta(0);
  return chart.hasSingleBarIndexLayout(meta.controller!, meta.iScale!);
}

Object? _plain(Object? value) {
  if (value is double && !value.isFinite) return value.toString();
  if (value is List) return value.map(_plain).toList();
  if (value is JsObjectFace) {
    return {for (final key in value.jsKeys()) key: _plain(value.jsRead(key))};
  }
  return value;
}

Object _observe(Chart chart, int datasets) => [
  for (var d = 0; d < datasets; d++)
    [
      for (final e in chart.getDatasetMeta(d).data)
        [
          for (final key in [
            'x',
            'y',
            'base',
            'width',
            'height',
            'horizontal',
            'enableBorderRadius',
            'borderSkipped',
            'inflateAmount',
          ])
            _plain((e as types.ElementPropertyRead)[key]),
        ],
    ],
];

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  Chart.register([
    BarController.jsClass,
    BarElement.jsClass,
    CategoryScale.jsClass,
    LinearScale.jsClass,
  ]);

  for (final horizontal in [false, true]) {
    test(
      'live single dataset geometry across changes, horizontal=$horizontal',
      () {
        final ca = _config(horizontal: horizontal);
        final cb = _config(horizontal: horizontal);
        final a = _chart(ca);
        final b = _chart(cb, reference: true);
        expect(_eligible(a), isTrue);
        for (var step = 0; step < 6; step++) {
          for (final cfg in [ca, cb]) {
            _dataset(cfg)['data'] = <double>[4.0 + step, 2, 0, -1];
            if (step == 1) _dataset(cfg)['stack'] = 'one';
            if (step == 2) _dataset(cfg)['grouped'] = false;
            if (step == 3) _dataset(cfg)['barThickness'] = 'flex';
            if (step == 4) _dataset(cfg)['minBarLength'] = 12.0;
          }
          if (step == 4) {
            a.resize(420, 300);
            b.resize(420, 300);
          }
          if (step == 5) {
            a.toggleDataVisibility(1);
            b.toggleDataVisibility(1);
          }
          a.update('none');
          b.update('none');
          expect(_observe(a, 1), _observe(b, 1), reason: 'step $step');
        }
        for (final cfg in [ca, cb]) {
          _datasets(cfg).add(<String, Object?>{
            'data': <double>[2, 4, 1, 3],
          });
        }
        a.update('none');
        b.update('none');
        expect(_eligible(a), isFalse);
        expect(_observe(a, 2), _observe(b, 2));
        for (final cfg in [ca, cb]) {
          _datasets(cfg).removeLast();
        }
        a.update('none');
        b.update('none');
        expect(_eligible(a), isTrue);
        expect(_observe(a, 1), _observe(b, 1));
      },
    );
  }

  test('typed root and dataset scopes qualify, routed ids do not', () {
    final dataset = BarControllerDatasetOptionsScope()
      ..data = <double>[1, 2, 3, 4];
    final data = ChartDataScope()
      ..labels = <String>['a', 'b', 'c', 'd']
      ..datasets = <Object?>[dataset];
    final root = ChartConfigurationScope()
      ..type = 'bar'
      ..data = data
      ..options = (ChartOptionsScope()
        ..responsive = false
        ..animation = false);
    final chart = _chart(root);
    expect(_eligible(chart), isTrue);
    final source = SingleBarDatasetSource(root);
    expect(source.read('x'), isTrue);
    var reads = 0;
    dataset.addRoute(
      'xAxisID',
      ScopeRoute(
        (_) {
          reads++;
          return 'x';
        },
        targetScope: null,
        targetName: 'x',
      ),
    );
    expect(source.read('x'), isFalse);
    expect(reads, 0);
  });

  test(
    'container identities are cached, contents and class changes stay live',
    () {
      final cfg = _config();
      final source = SingleBarDatasetSource(cfg);
      expect(source.read('x'), isTrue);
      _dataset(cfg)['xAxisID'] = null;
      expect(source.read('x'), isFalse);
      _dataset(cfg).remove('xAxisID');
      expect(source.read('x'), isTrue);
      _datasets(cfg).add(<String, Object?>{});
      expect(source.read('x'), isFalse);
      _datasets(cfg).removeLast();
      expect(source.read('x'), isTrue);
      final list = _ReadList();
      (cfg['data']! as Map<String, Object?>)['datasets'] = list;
      expect(source.read('x'), isFalse);
      expect(list.reads, 0);
      cfg['data'] = <String, Object?>{
        'datasets': <Object?>[<String, Object?>{}],
      };
      expect(source.read('x'), isTrue);
      final reads = <String>[];
      cfg['data'] = _ReadMap(<String, Object?>{}, reads);
      expect(source.read('x'), isFalse);
      expect(reads, isEmpty);
    },
  );

  test(
    'scale key order cache sees same-length replacement and numeric ids',
    () {
      final cache = BarScaleKeys();
      final map = <String, int>{'x': 1, '10': 2, '2': 3};
      expect(cache.read(map), ['2', '10', 'x']);
      map.remove('2');
      map['1'] = 4;
      expect(cache.read(map), ['1', '10', 'x']);
      map.clear();
      expect(cache.read(map), isEmpty);
    },
  );

  test('indexed stacks and custom scales retain their original calls', () {
    Chart.register([_ObservedCategory.value]);
    addTearDown(() => Chart.unregister([_ObservedCategory.value]));
    final ca = _config(skipNull: true);
    final cb = _config(skipNull: true);
    for (final cfg in [ca, cb]) {
      (_scales(cfg)['x']! as Map<String, Object?>)['type'] =
          'observed-bar-category';
    }
    final a = _chart(ca);
    final b = _chart(cb, reference: true);
    expect(_eligible(a), isFalse);
    final sa = a.scales['x']! as _ObservedCategory;
    final sb = b.scales['x']! as _ObservedCategory;
    sa.matches = 0;
    sb.matches = 0;
    a.update('none');
    b.update('none');
    expect(sa.matches, greaterThan(0));
    expect(sa.matches, sb.matches);
    expect(_observe(a, 1), _observe(b, 1));
  });

  test(
    'a callback between the two topology observations can add a dataset',
    () {
      final charts = <Chart>[];
      final logs = <List<String>>[];
      for (final reference in [false, true]) {
        final cfg = _config();
        final log = <String>[];
        var armed = false;
        var reads = 0;
        _dataset(cfg)['xAxisID'] = 'x';
        _scales(cfg)['0'] = <String, Object?>{
          'type': 'category',
          'axis': 'x',
          'display': false,
        };
        double categoryPercentage(Object? context) {
          if (armed) {
            log.add('category');
            reads++;
            if (reads == 2) {
              _datasets(cfg).add(<String, Object?>{
                'xAxisID': '0',
                'data': <double>[1, 2, 3, 4],
              });
            }
          }
          return 0.8;
        }

        _dataset(cfg)['categoryPercentage'] = categoryPercentage;
        double maxBarThickness(Object? context) {
          if (armed) {
            log.add('max');
            final chart = jsGet(context, 'chart')! as Chart;
            chart.options.readKey(OptionKey.indexAxis);
            final callback = _dataset(cfg)['categoryPercentage'];
            chart
                .getDatasetMeta(0)
                .controller!
                .options
                .jsDelete('categoryPercentage');
            _dataset(cfg)['categoryPercentage'] = callback;
          }
          return 100;
        }

        _dataset(cfg)['maxBarThickness'] = maxBarThickness;
        final chart = _chart(cfg, reference: reference);
        armed = true;
        chart.update('none');
        expect(_datasets(cfg), hasLength(2));
        charts.add(chart);
        logs.add(log);
      }
      expect(logs[0], logs[1]);
      expect(_observe(charts[0], 1), _observe(charts[1], 1));
    },
  );

  test('reentrant update from a range callback preserves outer geometry', () {
    final charts = <Chart>[];
    final logs = <List<String>>[];
    for (final reference in [false, true]) {
      final cfg = _config();
      var armed = false;
      var busy = false;
      var reads = 0;
      final log = <String>[];
      double categoryPercentage(Object? context) {
        if (armed && !busy) {
          reads++;
          log.add('category');
          if (reads == 2) {
            busy = true;
            log.add('reenter');
            final chart = jsGet(context, 'chart')! as Chart;
            _dataset(cfg)['data'] = <double>[4, 3, 2, 1];
            chart.update('none');
            busy = false;
          }
        }
        return 0.8;
      }

      _dataset(cfg)['categoryPercentage'] = categoryPercentage;
      double maxBarThickness(Object? context) {
        if (armed && !busy) {
          final chart = jsGet(context, 'chart')! as Chart;
          final callback = _dataset(cfg)['categoryPercentage'];
          chart
              .getDatasetMeta(0)
              .controller!
              .options
              .jsDelete('categoryPercentage');
          _dataset(cfg)['categoryPercentage'] = callback;
        }
        return 100;
      }

      _dataset(cfg)['maxBarThickness'] = maxBarThickness;
      final chart = _chart(cfg, reference: reference);
      armed = true;
      chart.update('none');
      expect(log, contains('reenter'));
      charts.add(chart);
      logs.add(log);
    }
    expect(logs[0], logs[1]);
    expect(_observe(charts[0], 1), _observe(charts[1], 1));
  });

  test('raw dataset getters retain their reads when the proof fails', () {
    final charts = <Chart>[];
    final traces = <List<String>>[];
    for (final reference in [false, true]) {
      final cfg = _config();
      final reads = <String>[];
      _datasets(cfg)[0] = _ReadMap(_dataset(cfg), reads);
      final chart = _chart(cfg, reference: reference);
      reads.clear();
      chart.update('none');
      expect(_eligible(chart), isFalse);
      charts.add(chart);
      traces.add(reads);
    }
    expect(traces[0], traces[1]);
    expect(_observe(charts[0], 1), _observe(charts[1], 1));
  });

  test(
    'ordered-value shortcut matches the original stable sort and uniqueness',
    () {
      Chart.register([_ValuesController.value]);
      addTearDown(() => Chart.unregister([_ValuesController.value]));
      for (final values in <List<double>>[
        [0, 1, 2, 3],
        [0, 0, 1, 1, 2],
        [3, 1, 2, 0],
        [-0.0, 0, 1],
        [double.nan, 2, 1],
        [double.infinity, 1, double.negativeInfinity],
      ]) {
        _ValuesController.values = values;
        final cfg = _config()..['type'] = 'bar-values-probe';
        final chart = _chart(cfg);
        final expected = arrayUnique(
          jsSort(List<double>.of(values), (a, b) => a - b),
        );
        final actual = chart.scales['x']!.cache.bar!;
        expect(_plain(actual), _plain(expected));
        expect(
          [for (final v in actual) v == 0 && v.isNegative],
          [for (final v in expected) v == 0 && v.isNegative],
        );
      }
    },
  );
  test('cached trait probes never resolve a scriptable option', () {
    var calls = 0;
    final raw = <String, Object?>{
      'categoryPercentage': (Object? _) {
        calls++;
        return 0.8;
      },
      'barPercentage': 0.9,
    };
    final view = attachContext(createResolver([raw]), <String, Object?>{});
    expect(view.hasCachedStaticBarTraits, isFalse);
    expect(calls, 0);
    view
      ..readKey(OptionKey.barThickness)
      ..readKey(OptionKey.categoryPercentage)
      ..readKey(OptionKey.barPercentage);
    expect(view.hasCachedStaticBarTraits, isTrue);
    expect(calls, 1);
    view.jsDelete('categoryPercentage');
    expect(view.hasCachedStaticBarTraits, isFalse);
    expect(calls, 1);
  });

  test('bar literal constructor preserves present bits and key order', () {
    final actual = BarProperties.geometry(
      horizontal: false,
      base: 10,
      enableBorderRadius: true,
      x: 2,
      y: 3,
      height: 7,
      width: 4,
    );
    final expected = BarProperties()
      ..horizontal = false
      ..base = 10
      ..enableBorderRadius = true
      ..x = 2
      ..y = 3
      ..height = 7
      ..width = 4;
    void same() {
      expect(actual.jsKeys(), expected.jsKeys());
      expect(_plain(actual), _plain(expected));
      for (var i = 0; i < 10; i++) {
        expect(actual.hasSlot(i), expected.hasSlot(i));
        expect(actual.isForeignSlot(i), expected.isForeignSlot(i));
      }
    }

    same();
    for (final p in [actual, expected]) {
      p
        ..options = <String, Object?>{'shared': true}
        ..borderSkipped = (BorderSkipped()..bottom = true)
        ..inflateAmount = 0
        ..jsDelete('x')
        ..jsWrite('extra', 9)
        ..jsWrite('x', '12')
        ..jsWrite('2', 8);
    }
    same();
    actual.x = 4;
    expected.x = 4;
    same();
  });
}
