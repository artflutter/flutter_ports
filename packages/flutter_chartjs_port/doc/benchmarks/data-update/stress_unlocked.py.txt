import subprocess,time,os,signal
from pathlib import Path
app='/tmp/chart-data-update.app'
bundle='com.artflutter.dataupdate.benchmark'
for call in ['full','data']:
 path=Path('/tmp/data-update-stress-unlocked-'+call+'.log')
 if path.exists():raise RuntimeError('Result already exists: '+str(path))
 path.write_text('')
 subprocess.run(['open','-n','-a',app,'--env','MODE=stress','--env','CALL='+call,'--stdout',str(path),'--stderr',str(path)],check=True)
 pid=None
 try:
  time.sleep(2)
  code='ObjC.import("AppKit"); var apps=$.NSRunningApplication.runningApplicationsWithBundleIdentifier("'+bundle+'"); console.log(apps.count ? apps.objectAtIndex(0).processIdentifier : 0);'
  result=subprocess.run(['osascript','-l','JavaScript','-e',code],capture_output=True,text=True,check=True)
  pid=int(result.stderr.strip())
  if not pid:raise RuntimeError('No benchmark process')
  for i in range(180):
   if i%12==0:
    code='ObjC.import("AppKit"); var app=$.NSRunningApplication.runningApplicationWithProcessIdentifier('+str(pid)+'); console.log(app.active);'
    result=subprocess.run(['osascript','-l','JavaScript','-e',code],capture_output=True,text=True,check=True)
    print('active',call,i,result.stderr.strip(),flush=True)
   txt=path.read_text()
   if 'STRESS {' in txt:
    print(call,next(l for l in txt.splitlines() if 'STRESS {' in l),flush=True);break
   time.sleep(.25)
  else:raise RuntimeError('No summary: '+txt)
 finally:
  if pid:os.kill(pid,signal.SIGTERM)
