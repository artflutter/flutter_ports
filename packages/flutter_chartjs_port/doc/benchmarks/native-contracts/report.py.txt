from pathlib import Path
import json,statistics,hashlib,gzip,shutil

runs=json.loads((w/'comparisons.json').read_text());comparison=[]

   comparison.append({'chart':chart,'style':style,'connected':connected,'beforeMeanUiMs':before,'afterMeanUiMs':after,'changePct':100*(after/before-1),'beforeUiP90Ms':[r['buildP90Ms'] for r in a],'afterUiP90Ms':[r['buildP90Ms'] for r in b]})
