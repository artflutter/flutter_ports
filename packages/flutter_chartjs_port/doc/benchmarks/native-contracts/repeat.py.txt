from pathlib import Path
import os,json,subprocess
out=Path(__file__).resolve().parent

def run(label,variant,extra,expected):
 cached=out/(label+'.json')
 if cached.exists():
  print('REUSE VALIDATED',label,flush=True);return json.loads(cached.read_text())
 print('START',label,flush=True);result=out/(label+'.results');result.unlink(missing_ok=True);Path(str(result)+'.window').unlink(missing_ok=True)
 env={**os.environ,'BENCH_DISPLAY_ID':'1','FRAMES':'1','PIXELS':'0','N':'20000','SHAPE':'plain','CALLS':'update','STYLES':'indexed','CONNECTED':'0,1','LABELS':'200','SPIN':'0','SECONDS':'8','OUTPUT':str(result),**extra}
 with (out/(label+'.log')).open('w') as log:subprocess.run([f'/tmp/native-contracts-{variant}.app/Contents/MacOS/chartjs_gallery','-ApplePersistenceIgnoreState','YES'],env=env,stdout=log,stderr=subprocess.STDOUT,timeout=900,check=True)
 rows=[json.loads(l.split(' ',1)[1]) for l in result.read_text().splitlines() if l.startswith('LINE_FRAME ')]
 assert len(rows)==expected and all(r['samples']>=8 and r['frameTimings']>=50 and r['items']==20000 and r['canvas']=='1036.0x520.0@2.0' for r in rows),(label,'invalid rows')
 vis=[json.loads(l) for l in Path(str(result)+'.window').read_text().splitlines()];prepared=next(r for r in vis if r['event']=='prepared');assert (prepared['width'],prepared['height'],prepared['dpr'],prepared['displayId'],prepared['maxFps'])==(1100,900,2,1,120)
 assert not any(r['time']>prepared['time']+1 and (r['occluded'] or r['appHidden'] or r['miniaturized'] or (r['width'],r['height'],r['dpr'],r['displayId'],r['maxFps'])!=(1100,900,2,1,120)) for r in vis),(label,'window changed')
 (out/(label+'.json')).write_text(json.dumps(rows,indent=2)+'\n');print('DONE',label,flush=True);return rows
rows={}
for tag,variant in [('a1','baseline'),('b1','production'),('b2','production'),('a2','baseline')]:
 rows[tag]=run('line-indexed-repeat-'+tag,variant,{'TYPE':'line','STYLES':'indexed','CONNECTED':'1','CALLS':'update','SECONDS':'12'},1)
 (out/'line-indexed-repeat.json').write_text(json.dumps(rows,indent=2)+'\n')
print('CONNECTED INDEXED REPEAT COMPLETE',flush=True)
