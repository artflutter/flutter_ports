// Temporary diagnostic of chart_benchmark_lab in native profile mode.
// MODE=bulk presses the real Capture all button.
// MODE=hover times each phase of real frames, with an optional mouse hover.
import 'dart:async';
import 'dart:convert';
import 'dart:io';
import 'dart:ui' as ui;

import 'package:chart_benchmark_lab/benchmark/config.dart';
import 'package:chart_benchmark_lab/benchmark/controller.dart';

import 'package:flutter/rendering.dart';
import 'package:flutter_chartjs_port/flutter_chartjs_port.dart' as cj;

final env = Platform.environment;
bool flag(String name) => env[name] == '1';
final watch = Stopwatch()..start();

Future<void> out(String line) async {
  stdout.writeln(
    'PROBE ${(watch.elapsedMilliseconds / 1000).toStringAsFixed(2)} $line',
  );
  await stdout.flush();
}

Element? findElement(bool Function(Element) test) {
  Element? found;
  void visit(Element e) {
    if (found != null) return;
    if (test(e)) {
      found = e;
      return;
    }
    e.visitChildren(visit);
  }

  WidgetsBinding.instance.rootElement?.visitChildren(visit);
  return found;
}

Offset centerOf(Element e) {
  final box = e.findRenderObject()! as RenderBox;
  return box.localToGlobal(box.size.center(Offset.zero));
}

Future<void> tapAt(Offset position) async {
  const device = 777;
  GestureBinding.instance
    ..handlePointerEvent(
      PointerDownEvent(
        kind: PointerDeviceKind.mouse,
        device: device,
        position: position,
      ),
    )
    ..handlePointerEvent(
      PointerUpEvent(
        kind: PointerDeviceKind.mouse,
        device: device,
        position: position,
      ),
    );
  await Future<void>.delayed(const Duration(milliseconds: 100));
}

class Phase {
  double totalUs = 0;
  double maxUs = 0;
  int count = 0;
  int? _start;
  void begin() => _start = watch.elapsedMicroseconds;
  void end() {
    final start = _start;
    if (start == null) return;
    final us = (watch.elapsedMicroseconds - start).toDouble();
    totalUs += us;
    if (us > maxUs) maxUs = us;
    count++;
    _start = null;
  }

  void reset() {
    totalUs = 0;
    maxUs = 0;
    count = 0;
  }

  String get text =>
      '${count == 0 ? 0 : (totalUs / count / 1000).toStringAsFixed(3)}ms'
      ' max=${(maxUs / 1000).toStringAsFixed(2)} n=$count';
}

final update = Phase();
final replay = Phase();
final draw = Phase();
final datasets = Phase();
final tooltipDraw = Phase();
var layouts = 0;
var skipNextDraw = false;
var skippedDraws = 0;
var framesWithActive = 0;
var framesDrawn = 0;

Map<String, Object?> timingPlugin() => <String, Object?>{
  'id': 'labProbeTiming',
  'supportsDataUpdates': true,
  'beforeUpdate': (Object? c, Object? a, Object? o) => update.begin(),
  'afterUpdate': (Object? c, Object? a, Object? o) => update.end(),
  'beforeLayout': (Object? c, Object? a, Object? o) => layouts++,
  'beforeEvent': (Object? c, Object? a, Object? o) => replay.begin(),
  'afterEvent': (Object? c, Object? a, Object? o) {
    replay.end();
    if (flag('DRAW_ONCE') && (a! as dynamic).replay == true) skipNextDraw = true;
  },
  'beforeDraw': (Object? c, Object? a, Object? o) {
    if (skipNextDraw) {
      skipNextDraw = false;
      skippedDraws++;
      return false;
    }
    draw.begin();
    return null;
  },
  'afterDraw': (Object? c, Object? a, Object? o) {
    draw.end();
    framesDrawn++;
    if ((c! as cj.Chart).getActiveElements().isNotEmpty) framesWithActive++;
  },
  'beforeDatasetsDraw': (Object? c, Object? a, Object? o) => datasets.begin(),
  'afterDatasetsDraw': (Object? c, Object? a, Object? o) => datasets.end(),
  'beforeTooltipDraw': (Object? c, Object? a, Object? o) => tooltipDraw.begin(),
  'afterTooltipDraw': (Object? c, Object? a, Object? o) => tooltipDraw.end(),
};

Future<void> finish(int code) async {
  await stdout.flush();
  exit(code);
}

Future<void> runHover(BenchmarkController c) async {
  final markers = flag('MARKERS');
  final hover = flag('HOVER');
  final tooltips = !flag('NO_TOOLTIPS');
  final count = int.parse(env['COUNT'] ?? '20000');
  await Future<void>.delayed(const Duration(milliseconds: 2500));
  c.configure(
    BenchmarkConfig(count: count, markers: markers, tooltips: tooltips),
  );
  await Future<void>.delayed(const Duration(milliseconds: 600));
  c.start(capture: true);
  // Starting a capture recreates the chart, so the hover must come after it.
  await Future<void>.delayed(const Duration(milliseconds: 600));
  if (hover) {
    final chart = findElement((e) => e.widget is cj.ChartView)!;
    final center = centerOf(chart);
    const device = 4242;
    GestureBinding.instance
      ..handlePointerEvent(
        PointerAddedEvent(
          kind: PointerDeviceKind.mouse,
          device: device,
          position: center,
        ),
      )
      ..handlePointerEvent(
        PointerHoverEvent(
          kind: PointerDeviceKind.mouse,
          device: device,
          position: center,
        ),
      );
  }
  if (hover && flag('MOVE')) {
    final center = centerOf(findElement((e) => e.widget is cj.ChartView)!);
    var step = 0;
    Timer.periodic(const Duration(milliseconds: 8), (_) {
      step++;
      GestureBinding.instance.handlePointerEvent(
        PointerHoverEvent(
          kind: PointerDeviceKind.mouse,
          device: 4242,
          position: center + Offset(((step ~/ 4) % 40) - 20.0, (step % 7) - 3.0),
        ),
      );
    });
  }
  await Future<void>.delayed(const Duration(milliseconds: 1400));
  for (final p in [update, replay, draw, datasets, tooltipDraw]) {
    p.reset();
  }
  layouts = 0;
  framesWithActive = 0;
  framesDrawn = 0;
  while (c.controlsLocked) {
    await Future<void>.delayed(const Duration(milliseconds: 100));
  }
  final m = c.results.isEmpty ? null : c.results.first.metrics;
  await out(
    'RESULT markers=$markers hover=$hover tooltips=$tooltips count=$count '
    'view=${c.viewport.width}x${c.viewport.height} hz=${c.refreshRate} '
    'fps=${m?.fps.toStringAsFixed(1)} '
    'frameP90=${m?.intervalP90.toStringAsFixed(2)} '
    'uiP90=${m?.uiP90.toStringAsFixed(2)} '
    'rasterP90=${m?.rasterP90.toStringAsFixed(2)} '
    'over=${m?.overBudgetPercent.toStringAsFixed(1)} msg="${c.message}"',
  );
  await out('  update  ${update.text}');
  await out('  replay  ${replay.text}');
  await out('  draw    ${draw.text}');
  await out('  datasets ${datasets.text}');
  await out('  tooltip ${tooltipDraw.text}');
  await out(
    '  layouts=$layouts drawn=$framesDrawn withActive=$framesWithActive skipped=$skippedDraws',
  );
  await finish(0);
}

Future<void> runBulk(BenchmarkController c, GlobalKey boundary) async {
  var last = '';
  void report() {
    final now =
        'lib=${c.library.name} pos=${c.batchPosition} '
        'batch=${c.batchCapturing} run=${c.running} cap=${c.capturing} '
        'fin=${c.finishing} results=${c.results.length} '
        'match=${c.results.where((r) => r.comparisonKey == c.comparisonKey).length} '
        'view=${c.viewport.width}x${c.viewport.height} dpr=${c.dpr} '
        'hz=${c.refreshRate} visible=${c.windowVisible} msg="${c.message}"';
    if (now != last) {
      last = now;
      unawaited(out(now));
    }
  }

  c.addListener(report);
  await Future<void>.delayed(const Duration(milliseconds: 2500));
  if (env['COUNT'] != null) {
    c.configure(
      BenchmarkConfig(
        count: int.parse(env['COUNT']!),
        markers: flag('MARKERS'),
        tooltips: !flag('NO_TOOLTIPS'),
      ),
    );
  }
  await Future<void>.delayed(const Duration(milliseconds: 500));
  report();
  final button = findElement(
    (e) => e.widget.key == const ValueKey('capture.all'),
  )!;
  await out('TAP capture.all at ${centerOf(button)}');
  await tapAt(centerOf(button));
  final limit = Stopwatch()..start();
  while (limit.elapsed.inSeconds < 3 && !c.batchCapturing) {
    await Future<void>.delayed(const Duration(milliseconds: 50));
  }
  await out('batchCapturing after tap=${c.batchCapturing}');
  while (c.controlsLocked && limit.elapsed.inSeconds < 90) {
    await Future<void>.delayed(const Duration(milliseconds: 250));
    report();
  }
  await Future<void>.delayed(const Duration(milliseconds: 800));
  report();
  final table = findElement((e) => e.widget is DataTable);
  final rows = table == null ? -1 : (table.widget as DataTable).rows.length;
  final order = c.results.reversed.map((r) => r.library.name).toList();
  final samples = c.results.reversed.map((r) => r.metrics.samples).toList();
  await out('END savedOrder=$order samples=$samples visibleTableRows=$rows');
  await out('JSON ${jsonEncode(c.results.reversed.map((r) => {'lib': r.library.name, 'fps': r.metrics.fps}).toList())}');
  final render =
      boundary.currentContext!.findRenderObject()! as RenderRepaintBoundary;
  final image = await render.toImage();
  final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
  final path = '${Directory.systemTemp.path}/lab_probe_bulk.png';
  File(path).writeAsBytesSync(bytes!.buffer.asUint8List());
  await out('SCREENSHOT $path');
  await finish(0);
}

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  cj.Chart.register(cj.registerables);
  final mode = env['MODE'] ?? 'hover';
  if (mode == 'hover') cj.Chart.register([timingPlugin()]);
  FlutterError.onError = (d) {
    unawaited(out('FLUTTER_ERROR ${d.exceptionAsString()}'));
  };
  final boundary = GlobalKey();
  runApp(
    RepaintBoundary(
      key: boundary,
      child: ChartBenchmarkApp(
        onControllerReady: (c) {
          unawaited(mode == 'hover' ? runHover(c) : runBulk(c, boundary));
        },
      ),
    ),
  );
}
