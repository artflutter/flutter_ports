// Original code, not ported from Chart.js: isolated native pixel equivalence.
import 'dart:typed_data';
import 'dart:io';
import 'package:crypto/crypto.dart';
import 'package:flutter/material.dart';
import 'dart:convert';
import 'dart:ui' as ui;

import 'package:flutter_chartjs_port/src/core/color_estimates.dart';
import 'package:flutter_chartjs_port/src/canvas/canvas_ui.dart';
import 'package:flutter_chartjs_port/src/elements/element.point.dart';
import 'package:flutter_chartjs_port/src/elements/element_options.dart';

Future<Uint8List> render(
  int variant,
  int dpr,
  double radius,
  double width,
  int alpha,
  bool indexed,
  int transform,
) async {
  estimateAtlasTrace.clear();
  estimateDrawing = variant == 1 || variant == 3;
  estimatePacking = variant == 2 || variant == 3;
  final element = UiCanvasElement(width: 128 * dpr, height: 128 * dpr);
  final ctx = element.getContext2d();
  ctx.scale(dpr.toDouble(), dpr.toDouble());
  if (transform == 1) {
    ctx.translate(20, 10);
    ctx.rotate(.3);
  }
  if (transform == 2) {
    ctx.scale(1.2, .8);
  }
  for (var i = 0; i < 600; i++) {
    final point = PointElement()
      ..x = -5 + (i * 7.371) % 140
      ..y = -3 + (i * 5.127) % 136;
    point.options = PointOptions(<String, Object?>{
      'radius': radius,
      'hoverRadius': radius,
      'borderWidth': width,
      'pointStyle': 'circle',
      'rotation': 0.0,
      'backgroundColor': ui.Color.fromARGB(
        alpha,
        indexed && i.isOdd ? 255 : 30,
        90,
        200,
      ),
      'borderColor': const ui.Color(0xff2499ff),
    });
    point.draw(ctx);
  }
  // The fast entry must retain the last path and the observable paint state.
  ctx.fill();
  ctx.stroke();
  final recorder = ui.PictureRecorder();
  ctx.paint(ui.Canvas(recorder), ui.Offset.zero, 1, 1);
  final picture = recorder.endRecording();
  final image = picture.toImageSync(128 * dpr, 128 * dpr);
  final bytes = (await image.toByteData(format: ui.ImageByteFormat.rawRgba))!
      .buffer
      .asUint8List();
  image.dispose();
  picture.dispose();
  element.width = element.width;
  return bytes;
}


void main() {
 WidgetsFlutterBinding.ensureInitialized();
 var firstFrame = false;
 WidgetsBinding.instance.addPostFrameCallback((_) {firstFrame=true;});
 runApp(const MaterialApp(home:Scaffold(body:Text('Cold rendering validation'))));
 Future<void> run() async {
  final variant=Platform.environment['DRAW']=='1'?1:0;
  estimateCheckPixels=true;
  final rows=<Object>[];
  for(var i=0;i<6;i++) {
   final before=firstFrame;
   final pixels=await render(variant,1,1,.5,64,false,0);
   rows.add({'iteration':i,'variant':variant,'frameBefore':before,'frameAfter':firstFrame,
     'pixels':sha256.convert(pixels).toString(),
     'commands':sha256.convert(utf8.encode(jsonEncode(estimateAtlasTrace))).toString()});
   if(i==1)await Future<void>.delayed(const Duration(milliseconds:250));
  }
  File(Platform.environment['OUTPUT']!).writeAsStringSync(jsonEncode(rows));
  exit(0);
 }
 run().catchError((Object e,StackTrace s){print('$e\n$s');exit(1);});
}
