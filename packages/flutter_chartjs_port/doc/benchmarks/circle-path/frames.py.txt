import json,os,subprocess
from pathlib import Path
out=Path(__file__).resolve().parent

def run(label,app,extra,expected):
 print('START',label,flush=True)
 result=out/(label+'.results');result.unlink(missing_ok=True);Path(str(result)+'.window').unlink(missing_ok=True)
 env={**os.environ,'FRAMES':'1','PIXELS':'0','CALLS':'update','STYLES':'scriptable','CONNECTED':'0,1','LABELS':'0,200','SPIN':'1','SECONDS':'8',**extra,'OUTPUT':str(result)}
 with (out/(label+'.log')).open('w') as log:
  subprocess.run([f'/tmp/circle-path-{app}.app/Contents/MacOS/chartjs_gallery','-ApplePersistenceIgnoreState','YES'],env=env,stdout=log,stderr=subprocess.STDOUT,timeout=180,check=True)
 rows=[json.loads(l.split(' ',1)[1]) for l in result.read_text().splitlines() if l.startswith('LINE_FRAME ')]
 assert len(rows)==expected,(label,len(rows))
 assert all(r['samples']>=4 and r['frameTimings']>=25 and r['canvas']=='1036.0x520.0@2.0' for r in rows),(label,'invalid timings')
 visibility=[json.loads(l) for l in Path(str(result)+'.window').read_text().splitlines()]
 prepared=next(r for r in visibility if r['event']=='prepared')
 assert (prepared['width'],prepared['height'],prepared['dpr'])==(1100,900,2)
 assert not any(r['time']>prepared['time']+1 and (r['occluded'] or r['appHidden'] or r['miniaturized'] or r['width']!=1100 or r['height']!=900 or r['dpr']!=2) for r in visibility),(label,'visibility interrupted')
 (out/(label+'.json')).write_text(json.dumps(rows,indent=2)+'\n');print('DONE',label,len(rows),flush=True)
 return rows

paired={'baseline':[],'production':[]}
for call in ['draw','update','animate']:
 for tag,app in [('a','baseline'),('b','production')]:
  paired[app]+=run(f'frames-{tag}-{call}-scriptable',app,{'CALLS':call},4)
for style in ['static','indexed']:
 for call in ['update','animate']:
  for tag,app in [('a','baseline'),('b','production')]:
   paired[app]+=run(f'frames-{tag}-{call}-{style}',app,{'CALLS':call,'STYLES':style,'LABELS':'200'},2)
(out/'paired-frames.json').write_text(json.dumps(paired,indent=2)+'\n')
repeat={}
for tag,app in [('a1','baseline'),('b1','production'),('b2','production'),('a2','baseline')]:
 repeat[tag]=run('repeat-'+tag,app,{'CALLS':'update,animate','CONNECTED':'1','LABELS':'200','SECONDS':'12'},2)
(out/'repeat-frames.json').write_text(json.dumps(repeat,indent=2)+'\n')
print('VERIFIED 40 paired frame cells and 8 repeated heavy cells',flush=True)
